-- 1. DATABASE CREATION

DROP DATABASE IF EXISTS smart_college;

CREATE DATABASE smart_college;

USE smart_college;


-- ============================================================
-- 2. CREATE DEPARTMENTS TABLE
-- ============================================================

CREATE TABLE departments (
    department_id INT PRIMARY KEY AUTO_INCREMENT,
    department_name VARCHAR(100) NOT NULL UNIQUE,
    hod_name VARCHAR(100),
    location VARCHAR(100)
);


-- ============================================================
-- 3. CREATE STUDENTS TABLE
-- ============================================================

CREATE TABLE students (
    student_id INT PRIMARY KEY AUTO_INCREMENT,
    student_name VARCHAR(100) NOT NULL,
    gender ENUM('Male', 'Female', 'Other') NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(15),
    department_id INT NOT NULL,
    admission_year YEAR NOT NULL,

    CONSTRAINT fk_student_department
        FOREIGN KEY (department_id)
        REFERENCES departments(department_id)
);


-- ============================================================
-- 4. CREATE FACULTY TABLE
-- ============================================================

CREATE TABLE faculty (
    faculty_id INT PRIMARY KEY AUTO_INCREMENT,
    faculty_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(15),
    department_id INT NOT NULL,
    designation VARCHAR(50),

    CONSTRAINT fk_faculty_department
        FOREIGN KEY (department_id)
        REFERENCES departments(department_id)
);


-- ============================================================
-- 5. CREATE COURSES TABLE
-- ============================================================

CREATE TABLE courses (
    course_id INT PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(100) NOT NULL,
    course_code VARCHAR(20) NOT NULL UNIQUE,
    credits INT NOT NULL,
    department_id INT NOT NULL,
    faculty_id INT NOT NULL,

    CONSTRAINT chk_course_credits
        CHECK (credits BETWEEN 1 AND 6),

    CONSTRAINT fk_course_department
        FOREIGN KEY (department_id)
        REFERENCES departments(department_id),

    CONSTRAINT fk_course_faculty
        FOREIGN KEY (faculty_id)
        REFERENCES faculty(faculty_id)
);


-- ============================================================
-- 6. CREATE ENROLLMENTS TABLE
-- ============================================================

CREATE TABLE enrollments (
    enrollment_id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL,
    course_id INT NOT NULL,
    semester INT NOT NULL,
    enrollment_date DATE NOT NULL,

    CONSTRAINT uq_student_course_semester
        UNIQUE (student_id, course_id, semester),

    CONSTRAINT fk_enrollment_student
        FOREIGN KEY (student_id)
        REFERENCES students(student_id),

    CONSTRAINT fk_enrollment_course
        FOREIGN KEY (course_id)
        REFERENCES courses(course_id)
);


-- ============================================================
-- 7. CREATE ATTENDANCE TABLE
-- ============================================================

CREATE TABLE attendance (
    attendance_id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL,
    course_id INT NOT NULL,
    total_classes INT NOT NULL,
    attended_classes INT NOT NULL,

    CONSTRAINT chk_attendance
        CHECK (
            total_classes >= 0
            AND attended_classes >= 0
            AND attended_classes <= total_classes
        ),

    CONSTRAINT uq_attendance_student_course
        UNIQUE (student_id, course_id),

    CONSTRAINT fk_attendance_student
        FOREIGN KEY (student_id)
        REFERENCES students(student_id),

    CONSTRAINT fk_attendance_course
        FOREIGN KEY (course_id)
        REFERENCES courses(course_id)
);


-- ============================================================
-- 8. CREATE EXAMS TABLE
-- ============================================================

CREATE TABLE exams (
    exam_id INT PRIMARY KEY AUTO_INCREMENT,
    exam_name VARCHAR(100) NOT NULL,
    course_id INT NOT NULL,
    semester INT NOT NULL,
    exam_date DATE NOT NULL,
    max_marks INT NOT NULL DEFAULT 100,

    CONSTRAINT chk_max_marks
        CHECK (max_marks > 0),

    CONSTRAINT fk_exam_course
        FOREIGN KEY (course_id)
        REFERENCES courses(course_id)
);


-- ============================================================
-- 9. CREATE MARKS TABLE
-- ============================================================

CREATE TABLE marks (
    mark_id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL,
    exam_id INT NOT NULL,
    marks_obtained DECIMAL(5,2) NOT NULL,

    CONSTRAINT uq_student_exam
        UNIQUE (student_id, exam_id),

    CONSTRAINT chk_marks
        CHECK (marks_obtained >= 0),

    CONSTRAINT fk_marks_student
        FOREIGN KEY (student_id)
        REFERENCES students(student_id),

    CONSTRAINT fk_marks_exam
        FOREIGN KEY (exam_id)
        REFERENCES exams(exam_id)
);

-- 18. CREATE ATTENDANCE ANALYTICS VIEW
-- ============================================================

CREATE VIEW attendance_analytics AS
SELECT
    s.student_id,
    s.student_name,
    c.course_name,
    a.total_classes,
    a.attended_classes,
    ROUND(
        (a.attended_classes / NULLIF(a.total_classes, 0)) * 100,
        2
    ) AS attendance_percentage
FROM attendance a
JOIN students s ON a.student_id = s.student_id
JOIN courses c ON a.course_id = c.course_id;


-- ============================================================
-- 19. CREATE STUDENT PERFORMANCE VIEW
-- ============================================================

CREATE VIEW student_performance AS
SELECT
    s.student_id,
    s.student_name,
    c.course_name,
    e.exam_name,
    m.marks_obtained,
    e.max_marks,
    ROUND((m.marks_obtained / e.max_marks) * 100, 2)
        AS percentage,
    CASE
        WHEN m.marks_obtained >= 90 THEN 'Excellent'
        WHEN m.marks_obtained >= 75 THEN 'Very Good'
        WHEN m.marks_obtained >= 60 THEN 'Good'
        WHEN m.marks_obtained >= 40 THEN 'Pass'
        ELSE 'Fail'
    END AS performance
FROM marks m
JOIN students s ON m.student_id = s.student_id
JOIN exams e ON m.exam_id = e.exam_id
JOIN courses c ON e.course_id = c.course_id;


-- ============================================================
-- 20. CREATE DEPARTMENT SUMMARY VIEW
-- ============================================================

CREATE VIEW department_summary AS
SELECT
    d.department_id,
    d.department_name,
    COUNT(DISTINCT s.student_id) AS total_students,
    COUNT(DISTINCT f.faculty_id) AS total_faculty,
    COUNT(DISTINCT c.course_id) AS total_courses
FROM departments d
LEFT JOIN students s ON d.department_id = s.department_id
LEFT JOIN faculty f ON d.department_id = f.department_id
LEFT JOIN courses c ON d.department_id = c.department_id
GROUP BY d.department_id, d.department_name;
