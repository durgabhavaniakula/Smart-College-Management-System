mysql> CREATE FUNCTION fn_student_attendance(p_student_id INT)
    -> RETURNS DECIMAL(5,2)
    -> READS SQL DATA
    -> BEGIN
    ->     DECLARE v_percentage DECIMAL(5,2);
    ->
    ->     SELECT COALESCE(
    ->         ROUND(SUM(attended_classes) * 100.0 /
    ->               NULLIF(SUM(total_classes), 0), 2),
    ->         0
    ->     )

    ->     FROM attendance

    ->     RETURN v_percentage;
